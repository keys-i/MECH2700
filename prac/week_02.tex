\section*{Question 1.}
\begin{solution}

  \begin{Section}{a.}
    The correlation is
    \[
      k(T) = AT^B e^{CT}e^{D / T},
    \]
    where
    \[
      A = 1.5531,\qquad B
        = 0.8703,\qquad C
        = -12.4882 \times 10^{-4},\qquad D
        = 143.6
    \]
    At \(T = 100K\),
    \[
      \begin{aligned}
        k(100) & = 1.5531(100)^{0.8703} e^{( - 12.4882 \times 10^{-4})(100)} e^{143.6 / 100} \\
               & = 317.11058019\ \mathrm{W/(mK)}
      \end{aligned}
    \]
    Similarly,
    \[
      \boxed{
            \begin{aligned}
                k(100)&=317.11058019\ \mathrm{W/(mK)} \\
                k(200)&=249.53883084\ \mathrm{W/(mK)} \\
                k(300)&=246.72527927\ \mathrm{W/(mK)}
            \end{aligned}}
    \]
  \end{Section}

  \begin{Section}{b.}
    The polynomial has the form
    \[
      k(T) = a T^2 + b T + c
    \]
    Substituting the three known points gives
    \[
      \begin{aligned}
        10000a + 100b + c & = 317.11058019 \\
        40000a + 200b + c & = 249.53883084 \\
        90000a + 300b + c & = 246.72527927
      \end{aligned}
    \]
    Subtracting the first equation from the second,
    \[
      30000a + 100b = -67.57174935
    \]
    Subtracting the second equation from the third,
    \[
      50000a + 100b = -2.81355157
    \]
    Subtracting these equations,
    \[
      20000a = 64.75819778
    \]
    so
    \[
      a = 0.003237909889
    \]
    Substituting this into the first reduced equation,
    \[
      30000(0.003237909889) + 100b = -67.57174935
    \]
    which gives
    \[
      b = -1.647090460
    \]
    Finally, using the first original equation,
    \[
      10000a + 100b + c = 317.11058019
    \]
    giving
    \[
      c = 449.4405273
    \]
    Therefore,
    \[
      k_{\mathrm{interp}}(T) = 0.003237909889 T^2 - 1.647090460 T + 449.4405273
    \]
  \end{Section}

  \begin{Section}{c.}
    \[
      \begin{aligned}
        k_{\mathrm{interp}}(250) & = 0.003237909889(250)^2 - 1.647090460(250) + 449.4405273 \\
                                 & = 240.03728033\ \mathrm{W/(mK)}
      \end{aligned}
    \]
    Thus,
    \[
      \boxed{k_{\mathrm{interp}}(250)=240.0373\ \mathrm{W/(mK)}}
    \]
  \end{Section}

  \begin{Section}{d.}
    The exact value from the original correlation is
    \[
      k_{\mathrm{exact}} = 246.60191848 \mathrm{W/(mK)}
    \]
    Therefore,
    \[
      \begin{aligned}
        \mathrm{error} & = \left| \frac{k_{\mathrm{interp}}-k_{\mathrm{exact}}} {k_{\mathrm{exact}}} \right| \times 100 \\
                       & = \left| \frac{240.03728033-246.60191848} {246.60191848} \right| \times 100 \\
                       & = 2.6620 \%
      \end{aligned}
    \]
    Thus,
    \[
      \boxed{\mathrm{Percentage\ error}\approx2.66\%}
    \]
  \end{Section}
\end{solution}
